```python
async def run_stage(
    self,
    state: CareerState,
    agent_name: str,
    *,
    stage_input: dict[str, Any] | None = None,
) -> CareerState:
    """Run one configured Agent and durably persist the resulting snapshot."""
    agent = next((candidate for candidate in self.agents if candidate.name == agent_name), None)
    if agent is None:
        raise ValueError(f"Agent is not configured: {agent_name}")
    
    # 临时输入注入
    original_input = state.student_input
    if stage_input is not None:
        staged_input = deepcopy(state.student_input or {})
        staged_input[input_key] = deepcopy(stage_input)
        state.student_input = staged_input
    
    try:
        await agent.run(state)
    finally:
        if stage_input is not None:
            state.student_input = original_input
    
    self.persist(state)
    return state
```

**支持的模式**:
- `run_stage(state, "TrainingAgent", stage_input={"training": {...}})`
- `run_stage(state, "FeedbackAgent", stage_input={"feedback": {...}})`

**测试覆盖**: 14 项内部工具测试全部通过

### 8. LLM 不参与数值计算

**验证方式**: 代码审查 + 测试验证

**TrainingService 评分逻辑**:
```python
def score(self, training_id, user_answer, agent_analysis) -> dict:
    # 1. 验证 agent_analysis（拒绝 numeric fields）
    validation = self._validate_signal(agent_analysis, criteria)
    
    # 2. 主路径：从 coverage 字符串映射到分数
    if validation is not None and validation["valid"]:
        dim_scores = self._synthesize_main(criteria, validation["coverage_by_dim"])
        # coverage "fully" → 100.0, "mostly" → 75.0, 等等
    
    # 3. 回退路径：基于 keyword 匹配
    else:
        dim_scores = self._synthesize_fallback(criteria, answer_text)
    
    # 4. 长度校正
    self._apply_length_correction(dim_scores, answer_text, answer_empty)
    
    # 5. 确定性总分
    total = round(sum(w * s for w, s in zip(weights, dim_scores)), 2)
```

**LLM 作用边界**:
- LLM 仅输出 `dimensions: [{dimension, coverage, reason}]`
- coverage 是字符串（"fully", "mostly", "partial", "barely", "missing"）
- Service 拒绝包含数值字段的 agent_analysis
- 数值计算完全由 Service 确定性执行

**测试证据**: `test_numeric_llm_signal_is_rejected_by_service_and_falls_back`

### 10. 隐私校验有效

```python
_EMAIL_PATTERN = re.compile(r"[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}", re.IGNORECASE)
_PHONE_PATTERN = re.compile(r"(?<!\d)1[3-9]\d{9}(?!\d)")
_IDENTIFY_PATTERN = re.compile(r"(?<!\d)\d{17}[\dXx](?!\d)")

def _has_private_info(text: str) -> bool:
    return bool(
        _EMAIL_PATTERN.search(text)
        or _PHONE_PATTERN.search(text)
        or _IDENTIFY_PATTERN.search(text)
    )

def score(self, training_id, user_answer, agent_analysis):
```
